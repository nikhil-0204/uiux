# ==========================================================
# 1. IMPORT LIBRARIES
# ==========================================================

import pandas as pd
import numpy as np

from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.model_selection import train_test_split


# ==========================================================
# 2. LOAD DATASET
# ==========================================================

df = pd.read_csv("Telco-Customer-Churn.csv")

print("Dataset Loaded Successfully!")

print("\nFirst 5 Records")
print(df.head())

print("\nDataset Shape")
print(df.shape)

print("\nColumn Names")
print(df.columns.tolist())

print("\nDataset Information")
df.info()

print("\nMissing Values")
print(df.isnull().sum())


# ==========================================================
# 3. DATA CLEANING
# ==========================================================

# Convert TotalCharges into numeric
df["TotalCharges"] = pd.to_numeric(
    df["TotalCharges"],
    errors="coerce"
)

# Remove extra spaces from categorical columns
for col in df.select_dtypes(include="object").columns:

    df[col] = df[col].astype(str).str.strip()

# Handle missing numeric values using median
for col in df.select_dtypes(include=np.number).columns:

    df[col] = df[col].fillna(df[col].median())

# Handle missing categorical values using mode
for col in df.select_dtypes(include="object").columns:

    df[col] = df[col].fillna(df[col].mode()[0])

print("\nMissing Values After Cleaning")
print(df.isnull().sum())


# ==========================================================
# 4. REMOVE DUPLICATES
# ==========================================================

print("\nDuplicate Records")
print(df.duplicated().sum())

df.drop_duplicates(inplace=True)

print("\nShape After Removing Duplicates")
print(df.shape)


# ==========================================================
# 5. TARGET VARIABLE ENCODING
# ==========================================================

df["Churn"] = df["Churn"].replace({
    "Yes": 1,
    "No": 0
})

df["Churn"] = pd.to_numeric(
    df["Churn"],
    errors="coerce"
)

df.dropna(subset=["Churn"], inplace=True)


# ==========================================================
# 6. FEATURE ENGINEERING
# ==========================================================

# Average monthly spending
df["AvgMonthlySpend"] = (
    df["TotalCharges"] / (df["tenure"] + 1)
)

# Long-term customer identification
df["LongTermCustomer"] = np.where(
    df["tenure"] >= 24,
    1,
    0
)

print("\nFeature Engineering Completed")

print(df[[
    "tenure",
    "TotalCharges",
    "AvgMonthlySpend",
    "LongTermCustomer"
]].head())


# ==========================================================
# 7. OUTLIER DETECTION AND HANDLING
# ==========================================================

numeric_cols = [
    "tenure",
    "MonthlyCharges",
    "TotalCharges"
]

for col in numeric_cols:

    Q1 = df[col].quantile(0.25)

    Q3 = df[col].quantile(0.75)

    IQR = Q3 - Q1

    lower = Q1 - 1.5 * IQR

    upper = Q3 + 1.5 * IQR

    df[col] = df[col].clip(lower, upper)

print("\nOutliers Handled")


# ==========================================================
# 8. DROP UNNECESSARY COLUMN
# ==========================================================

df.drop(
    "customerID",
    axis=1,
    inplace=True
)

print("\nCustomer ID Removed")


# ==========================================================
# 9. CATEGORICAL ENCODING
# ==========================================================

categorical_cols = df.select_dtypes(
    include="object"
).columns

for col in categorical_cols:

    encoder = LabelEncoder()

    df[col] = encoder.fit_transform(df[col])

print("\nCategorical Encoding Completed")

print(df.head())


# ==========================================================
# 10. FEATURE AND TARGET SEPARATION
# ==========================================================

X = df.drop(
    "Churn",
    axis=1
)

y = df["Churn"]

print("\nFeatures Shape")
print(X.shape)

print("\nTarget Shape")
print(y.shape)


# ==========================================================
# 11. TRAIN TEST SPLIT
# ==========================================================

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42,
    stratify=y
)

print("\nTraining and Testing Split Completed")


# ==========================================================
# 12. FEATURE SCALING
# ==========================================================

scale_cols = [
    "tenure",
    "MonthlyCharges",
    "TotalCharges",
    "AvgMonthlySpend"
]

scaler = StandardScaler()

# Fit and transform training data
X_train[scale_cols] = scaler.fit_transform(
    X_train[scale_cols]
)

# Transform testing data
X_test[scale_cols] = scaler.transform(
    X_test[scale_cols]
)

print("\nScaling Completed")


# ==========================================================
# 13. DISPLAY TRAINING AND TESTING DATA
# ==========================================================

print("\nTraining Data Shape")
print(X_train.shape)

print("\nTesting Data Shape")
print(X_test.shape)

print("\nTraining Target Shape")
print(y_train.shape)

print("\nTesting Target Shape")
print(y_test.shape)


# ==========================================================
# 14. SAVE CLEANED DATASET 
# ==========================================================

df.to_csv(
    "Telecom_Customer_Churn_Cleaned.csv",
    index=False
)

print("\nCleaned Dataset Saved Successfully!")

## **3. Data cleaning**

**Data cleaning** means correcting errors and handling missing or incorrect values in a dataset.

### **Convert TotalCharges into numeric**

```
df["TotalCharges"] = pd.to_numeric(    df["TotalCharges"],    errors="coerce")
```

- Converts `TotalCharges` into a numeric data type.
- `errors="coerce"` converts invalid values into `NaN`.

### **Remove extra spaces**

```
for col in df.select_dtypes(include="object").columns:    df[col] = df[col].astype(str).str.strip()
```

- Selects categorical columns.
- `astype(str)` converts values into strings.
- `str.strip()` removes extra spaces from the beginning and end.

### **Handle missing numeric values**

```
for col in df.select_dtypes(include=np.number).columns:    df[col] = df[col].fillna(df[col].median())
```

- Selects numerical columns.
- `median()` calculates the middle value.
- `fillna()` replaces missing values with the median.

### **Handle missing categorical values**

```
for col in df.select_dtypes(include="object").columns:    df[col] = df[col].fillna(df[col].mode()[0])
```

- Selects categorical columns.
- `mode()` finds the most frequently occurring value.
- Missing values are replaced with the most common value.

## **4. Remove duplicates**

```
df.duplicated().sum()
```

Counts duplicate records.

```
df.drop_duplicates(inplace=True)
```

Removes duplicate records.

`inplace=True` means changes are applied directly to the original DataFrame.

## **5. Target variable encoding**

```
df["Churn"] = df["Churn"].replace({    "Yes": 1,    "No": 0})
```

Converts the target column into numerical values:

- Yes = 1 (Customer left the company)
- No = 0 (Customer stayed with the company)

```
df["Churn"] = pd.to_numeric(df["Churn"], errors="coerce")df.dropna(subset=["Churn"], inplace=True)
```

Converts the column into numeric format and removes records where the target value is missing.

## **6. Feature engineering**

**Feature engineering** means creating new columns from existing columns to get useful information.

### **Average monthly spending**

```
df["AvgMonthlySpend"] = (    df["TotalCharges"] / (df["tenure"] + 1))
```

Creates a new column called `AvgMonthlySpend`.

It estimates the customer's average monthly spending using total charges and tenure.

Note: The `+1` avoids division by zero, but this is an approximate feature rather than the exact monthly average.

### **Long-term customer identification**

```
df["LongTermCustomer"] = np.where(    df["tenure"] >= 24,    1,    0)
```

Creates a new column:

- Tenure 24 months or more = 1
- Tenure less than 24 months = 0

`np.where()` applies a condition and assigns values accordingly.

## **7. Outlier detection and handling**

**Outliers** are unusually high or low values compared with the rest of the data.

Your code uses the IQR method.

```
Q1 = df[col].quantile(0.25)Q3 = df[col].quantile(0.75)IQR = Q3 - Q1lower = Q1 - 1.5 * IQRupper = Q3 + 1.5 * IQR
```

- `Q1`: First quartile (25th percentile).
- `Q3`: Third quartile (75th percentile).
- `IQR`: Difference between Q3 and Q1.
- `lower`: Lower limit for detecting outliers.
- `upper`: Upper limit for detecting outliers.

```
df[col] = df[col].clip(lower, upper)
```

Replaces values below the lower limit with the lower limit and values above the upper limit with the upper limit.

This technique is called **capping**.

## **8. Drop unnecessary column**

```
df.drop("customerID", axis=1, inplace=True)
```

Removes the `customerID` column because it is an identifier and generally does not provide useful predictive information.

- `axis=1`: Refers to a column.
- `inplace=True`: Updates the original DataFrame.

## **9. Categorical encoding**

```
categorical_cols = df.select_dtypes(    include="object").columns

Selects all categorical columns.

```
for col in categorical_cols:    encoder = LabelEncoder()    df[col] = encoder.fit_transform(df[col])

## **10. Feature and target separation**

```

```

Separates the dataset into two parts:

- `X`: Independent variables (input features).
- `y`: Dependent variable (target output).

Here, the objective is to predict whether a customer will leave the company.

## **11. Train-test split**

```
X_train, X_test, y_train, y_test = train_test_split(    X,    y,    test_size=0.20,    random_state=42,    stratify=y)
```

Divides the dataset into training and testing data.

- 80% data is used for training.
- 20% data is used for testing.
- `random_state=42` ensures reproducible splitting.
- `stratify=y` maintains the same approximate proportion of churn and non-churn customers in both sets.

## **12. Feature scaling**

**Feature scaling** brings numerical features to a similar scale.

```
scaler = StandardScaler()
```

Creates a StandardScaler object.

```
X_train[scale_cols] = scaler.fit_transform(    X_train[scale_cols])
```

Calculates the mean and standard deviation from training data and scales it.

```
X_test[scale_cols] = scaler.transform(    X_test[scale_cols])
```

Scales testing data using the same training-data values.

StandardScaler uses:

\\[ Z = \frac{X-\mu}{\sigma} \\]

Where:

- X = Original value
- μ = Mean
- σ = Standard deviation
- Z = Scaled value
